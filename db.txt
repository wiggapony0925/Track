-- ============================================================
-- 1. USER PROFILES (Syncs with Supabase Auth)
-- ============================================================
-- Stores user profile data from Apple Sign-In and other auth providers.
-- Links to Supabase Auth via the user's UUID.

CREATE TABLE IF NOT EXISTS profiles (
  id UUID REFERENCES auth.users ON DELETE CASCADE PRIMARY KEY,
  
  -- Apple Sign-In data
  apple_user_id TEXT UNIQUE,           -- Apple's unique user identifier
  email TEXT,                          -- Email (may be hidden by Apple)
  full_name TEXT,                      -- Full name from Apple Sign-In
  given_name TEXT,                     -- First name
  family_name TEXT,                    -- Last name
  
  -- App-specific data
  username TEXT UNIQUE,                -- Optional display name
  avatar_url TEXT,                     -- Profile picture URL
  
  -- Preferences
  preferred_theme TEXT DEFAULT 'system' CHECK (preferred_theme IN ('system', 'light', 'dark')),
  notifications_enabled BOOLEAN DEFAULT true,
  
  -- Metadata
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()),
  last_login_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);

-- Auto-update the updated_at timestamp
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = timezone('utc'::text, now());
    RETURN NEW;
END;
$$ language 'plpgsql';

CREATE TRIGGER update_profiles_updated_at
    BEFORE UPDATE ON profiles
    FOR EACH ROW
    EXECUTE FUNCTION update_updated_at_column();


-- ============================================================
-- 2. FAVORITES (Cross-device sync)
-- ============================================================
-- Stores user's favorite routes/stops for quick access.
-- Syncs between devices and persists offline via SwiftData.

CREATE TABLE IF NOT EXISTS favorites (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id UUID REFERENCES auth.users ON DELETE CASCADE NOT NULL,
  
  -- Transit info
  route_id TEXT NOT NULL,              -- e.g., "MTA NYCT_L", "B63"
  route_display_name TEXT NOT NULL,    -- e.g., "L", "B63"
  stop_id TEXT NOT NULL,               -- GTFS stop ID
  stop_name TEXT NOT NULL,             -- Human-readable stop name
  direction TEXT,                      -- e.g., "Uptown", "Downtown"
  destination TEXT,                    -- e.g., "8 Av", "Canarsie"
  mode TEXT NOT NULL CHECK (mode IN ('bus', 'subway', 'lirr')),
  
  -- Optional location for map display
  stop_lat DOUBLE PRECISION,
  stop_lon DOUBLE PRECISION,
  
  -- Display order (for user reordering)
  display_order INT DEFAULT 0,
  
  -- Metadata
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()),
  
  -- Ensure no duplicate favorites per user
  UNIQUE(user_id, route_id, stop_id, direction)
);

CREATE TRIGGER update_favorites_updated_at
    BEFORE UPDATE ON favorites
    FOR EACH ROW
    EXECUTE FUNCTION update_updated_at_column();


-- ============================================================
-- 3. ROUTE INTERACTIONS (Analytics - Global Popularity)
-- ============================================================
-- Tracks which routes users click/track for trending analytics.
-- Used by the backend to surface popular routes.

CREATE TABLE IF NOT EXISTS route_interactions (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  
  -- Can be NULL for anonymous users
  user_id UUID REFERENCES auth.users ON DELETE SET NULL,
  
  -- Route info
  route_id TEXT NOT NULL,              -- Full route ID
  route_display_name TEXT,             -- Display name
  mode TEXT NOT NULL CHECK (mode IN ('bus', 'subway', 'lirr')),
  
  -- Interaction type
  interaction_type TEXT DEFAULT 'click' CHECK (interaction_type IN ('click', 'track', 'favorite', 'share')),
  
  -- Location context (optional - for regional trends)
  latitude DOUBLE PRECISION,
  longitude DOUBLE PRECISION,
  
  -- Metadata
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);

-- Index for fast aggregation queries
CREATE INDEX IF NOT EXISTS idx_route_interactions_route_id ON route_interactions(route_id);
CREATE INDEX IF NOT EXISTS idx_route_interactions_created_at ON route_interactions(created_at);
CREATE INDEX IF NOT EXISTS idx_route_interactions_mode ON route_interactions(mode);


-- ============================================================
-- 4. SCHEDULES (Widget Activation Schedules)
-- ============================================================
-- Stores when users want their widgets to automatically show transit.
-- Syncs across devices.

CREATE TABLE IF NOT EXISTS schedules (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES auth.users ON DELETE CASCADE NOT NULL,
  
  -- Schedule config
  days_of_week INT[] NOT NULL,         -- 0=Sun, 1=Mon, ..., 6=Sat (e.g., {1,2,3,4,5} for weekdays)
  start_time TIME NOT NULL,            -- e.g., '08:30:00'
  duration_minutes INT DEFAULT 15,     -- How long the widget stays active
  
  -- Optional: tie to a specific route
  route_id TEXT,
  direction TEXT,
  
  -- State
  is_enabled BOOLEAN DEFAULT true,
  
  -- Metadata
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);

CREATE TRIGGER update_schedules_updated_at
    BEFORE UPDATE ON schedules
    FOR EACH ROW
    EXECUTE FUNCTION update_updated_at_column();


-- ============================================================
-- 5. COMMUTE PATTERNS (Smart Suggestions)
-- ============================================================
-- Stores learned commute patterns for the Smart Suggester.
-- Used to predict which route a user likely wants.

CREATE TABLE IF NOT EXISTS commute_patterns (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES auth.users ON DELETE CASCADE NOT NULL,
  
  -- Pattern data
  route_id TEXT NOT NULL,
  direction TEXT NOT NULL,
  start_latitude DOUBLE PRECISION NOT NULL,
  start_longitude DOUBLE PRECISION NOT NULL,
  destination_station_id TEXT NOT NULL,
  destination_name TEXT NOT NULL,
  
  -- Time context
  time_of_day INT NOT NULL CHECK (time_of_day >= 0 AND time_of_day <= 23),  -- Hour 0-23
  day_of_week INT NOT NULL CHECK (day_of_week >= 0 AND day_of_week <= 6),   -- 0=Sun, 6=Sat
  
  -- Frequency tracking
  frequency INT DEFAULT 1,
  last_used TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()),
  
  -- Metadata
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()),
  
  -- Unique constraint to avoid duplicates
  UNIQUE(user_id, route_id, direction, destination_station_id, time_of_day, day_of_week)
);

CREATE TRIGGER update_commute_patterns_updated_at
    BEFORE UPDATE ON commute_patterns
    FOR EACH ROW
    EXECUTE FUNCTION update_updated_at_column();


-- ============================================================
-- 6. TRIP LOGS (Delay Predictor Training Data)
-- ============================================================
-- Stores historical trip data for ML-based delay prediction.

CREATE TABLE IF NOT EXISTS trip_logs (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES auth.users ON DELETE SET NULL,
  
  -- Trip details
  route_id TEXT NOT NULL,
  origin_station_id TEXT NOT NULL,
  destination_station_id TEXT NOT NULL,
  
  -- Time context
  time_of_day INT NOT NULL CHECK (time_of_day >= 0 AND time_of_day <= 23),
  day_of_week INT NOT NULL CHECK (day_of_week >= 0 AND day_of_week <= 6),
  
  -- Weather (for delay correlation)
  weather_condition TEXT CHECK (weather_condition IN ('clear', 'rain', 'snow')),
  
  -- Timing
  mta_predicted_time TIMESTAMP WITH TIME ZONE NOT NULL,
  actual_arrival_time TIMESTAMP WITH TIME ZONE,
  delay_seconds INT DEFAULT 0,
  
  -- Metadata
  trip_date DATE DEFAULT CURRENT_DATE,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);

CREATE INDEX IF NOT EXISTS idx_trip_logs_route_id ON trip_logs(route_id);
CREATE INDEX IF NOT EXISTS idx_trip_logs_trip_date ON trip_logs(trip_date);


-- ============================================================
-- 7. ROW LEVEL SECURITY (RLS)
-- ============================================================
-- Ensures users can only access their own data.

-- Enable RLS on all tables with user data
ALTER TABLE profiles ENABLE ROW LEVEL SECURITY;
ALTER TABLE favorites ENABLE ROW LEVEL SECURITY;
ALTER TABLE schedules ENABLE ROW LEVEL SECURITY;
ALTER TABLE commute_patterns ENABLE ROW LEVEL SECURITY;
ALTER TABLE trip_logs ENABLE ROW LEVEL SECURITY;

-- route_interactions allows anonymous inserts but only user reads own data
ALTER TABLE route_interactions ENABLE ROW LEVEL SECURITY;

-- ============================================================
-- PROFILES POLICIES
-- ============================================================
CREATE POLICY "Users can view own profile" 
  ON profiles FOR SELECT 
  USING (auth.uid() = id);

CREATE POLICY "Users can insert own profile" 
  ON profiles FOR INSERT 
  WITH CHECK (auth.uid() = id);

CREATE POLICY "Users can update own profile" 
  ON profiles FOR UPDATE 
  USING (auth.uid() = id);

  USING (auth.uid() = user_id);

  USING (auth.uid() = user_id);

  ON schedules FOR ALL 
  USING (auth.uid() = user_id);

-- ============================================================

-- ============================================================

  ON commute_patterns FOR SELECT 
  USING (auth.uid() = user_id);

CREATE POLICY "Users can manage own commute patterns" 
  ON commute_patterns FOR ALL 
  USING (auth.uid() = user_id);

-- ============================================================
-- TRIP LOGS POLICIES
-- ============================================================

  ON trip_logs FOR SELECT 
  USING (auth.uid() = user_id);

  ON route_interactions FOR INSERT 
  WITH CHECK (true);

-- Users can only view their own interactions (or anonymous ones)
CREATE POLICY "Users can view own interactions" 
  ON route_interactions FOR SELECT 
  USING (auth.uid() = user_id OR user_id IS NULL);

-- 8. HELPER FUNCTIONS
-- ============================================================

-- Function to get most popular routes (for backend use)
CREATE OR REPLACE FUNCTION get_popular_routes(
  p_limit INT DEFAULT 10,
  p_mode TEXT DEFAULT NULL,
  p_days INT DEFAULT 7
)
RETURNS TABLE (
  route_id TEXT,
  route_display_name TEXT,
  mode TEXT,
  interaction_count BIGINT
) AS $$
BEGIN
  RETURN QUERY
  SELECT 
    ri.route_id,
    MAX(ri.route_display_name) as route_display_name,
    ri.mode,
    COUNT(*) as interaction_count
  FROM route_interactions ri
  WHERE ri.created_at >= NOW() - (p_days || ' days')::INTERVAL
    AND (p_mode IS NULL OR ri.mode = p_mode)
  GROUP BY ri.route_id, ri.mode
  ORDER BY interaction_count DESC
  LIMIT p_limit;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;


-- Function to auto-create profile on user signup
CREATE OR REPLACE FUNCTION handle_new_user()
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO public.profiles (id, email, created_at)
  VALUES (NEW.id, NEW.email, NOW())
  ON CONFLICT (id) DO NOTHING;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Trigger to auto-create profile
CREATE OR REPLACE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE FUNCTION handle_new_user();
