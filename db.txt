-- Auto-update the updated_at timestamp on any table
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = timezone('utc'::text, now());
    RETURN NEW;
END;
$$ language 'plpgsql';


-- ============================================================
-- 1. USER PROFILES (Syncs with Supabase Auth)
-- ============================================================
-- Stores user profile data from Apple Sign-In.
-- Links to Supabase Auth via the user's UUID.

CREATE TABLE IF NOT EXISTS profiles (
  id UUID REFERENCES auth.users ON DELETE CASCADE PRIMARY KEY,
  
  -- Apple Sign-In data (provided on first sign-in only)
  apple_user_id TEXT UNIQUE,           -- Apple's unique user identifier
  email TEXT,                          -- Email (may be hidden by Apple)
  full_name TEXT,                      -- Full name from Apple Sign-In
  given_name TEXT,                     -- First name
  family_name TEXT,                    -- Last name
  
  -- Profile customization
  username TEXT,                        -- Optional display name
  avatar_url TEXT,                      -- Optional avatar URL
  
  -- App preferences (synced across devices)
  preferred_theme TEXT DEFAULT 'system' CHECK (preferred_theme IN ('system', 'light', 'dark')),
  notifications_enabled BOOLEAN DEFAULT true,
  
  -- Metadata
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()),
  last_login_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);

-- Add columns if they don't exist (for existing databases)
DO $$ BEGIN
  ALTER TABLE profiles ADD COLUMN IF NOT EXISTS username TEXT;
  ALTER TABLE profiles ADD COLUMN IF NOT EXISTS avatar_url TEXT;
EXCEPTION WHEN OTHERS THEN NULL;
END $$;

-- Trigger for auto-updating updated_at (idempotent)
DROP TRIGGER IF EXISTS update_profiles_updated_at ON profiles;
CREATE TRIGGER update_profiles_updated_at
    BEFORE UPDATE ON profiles
    FOR EACH ROW
    EXECUTE FUNCTION update_updated_at_column();


-- ============================================================
-- 2. ROUTE INTERACTIONS (Analytics - What routes are popular)
-- ============================================================
-- Tracks which routes users click or track via Live Activity.
-- Used to show trending routes and popular routes in your area.

CREATE TABLE IF NOT EXISTS route_interactions (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  
  -- Can be NULL for anonymous users (Continue without account)
  user_id UUID REFERENCES auth.users ON DELETE SET NULL,
  
  -- Route info
  route_id TEXT NOT NULL,              -- Full route ID (e.g., "MTA NYCT_L")
  route_display_name TEXT,             -- Display name (e.g., "L")
  mode TEXT NOT NULL CHECK (mode IN ('bus', 'subway', 'lirr', 'mnr')),
  
  -- Interaction type: what did the user do?
  interaction_type TEXT DEFAULT 'click' CHECK (interaction_type IN ('click', 'track')),
  -- 'click' = user tapped to view route details
  -- 'track' = user started a Live Activity to track this route
  
  -- Location context (optional - for regional popularity trends)
  latitude DOUBLE PRECISION,
  longitude DOUBLE PRECISION,
  
  -- Metadata
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);

-- Indexes for fast analytics queries (idempotent with IF NOT EXISTS)
CREATE INDEX IF NOT EXISTS idx_route_interactions_route_id ON route_interactions(route_id);
CREATE INDEX IF NOT EXISTS idx_route_interactions_created_at ON route_interactions(created_at);
CREATE INDEX IF NOT EXISTS idx_route_interactions_mode ON route_interactions(mode);


-- ============================================================
-- 3. FAVORITES (Saved routes/stops for quick access)
-- ============================================================
-- Stores user's favorite routes and stops.
-- Referenced by SupabaseManager.fetchFavorites(), addFavorite(), removeFavorite().
-- Syncs across devices so favorites appear on all user's Apple devices.

CREATE TABLE IF NOT EXISTS favorites (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id UUID REFERENCES auth.users ON DELETE CASCADE NOT NULL,
  
  -- Route/stop info
  route_id TEXT NOT NULL,              -- Full route ID (e.g., "MTA NYCT_L", "LIRR_9")
  route_display_name TEXT NOT NULL,    -- Display name (e.g., "L", "Port Washington Branch")
  stop_id TEXT NOT NULL,               -- Stop/station ID
  stop_name TEXT NOT NULL,             -- Human-readable stop name
  direction TEXT,                      -- Direction (e.g., "N", "S", "Eastbound")
  destination TEXT,                    -- Terminal station name
  mode TEXT NOT NULL CHECK (mode IN ('bus', 'subway', 'lirr', 'mnr')),
  
  -- Stop coordinates (for map display without re-fetching)
  stop_lat DOUBLE PRECISION,
  stop_lon DOUBLE PRECISION,
  
  -- User ordering (drag to reorder favorites)
  display_order INT DEFAULT 0,
  
  -- Metadata
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()),
  
  -- Prevent duplicate favorites
  UNIQUE(user_id, route_id, stop_id, direction)
);

-- Trigger for auto-updating updated_at (idempotent)
DROP TRIGGER IF EXISTS update_favorites_updated_at ON favorites;
CREATE TRIGGER update_favorites_updated_at
    BEFORE UPDATE ON favorites
    FOR EACH ROW
    EXECUTE FUNCTION update_updated_at_column();

CREATE INDEX IF NOT EXISTS idx_favorites_user_id ON favorites(user_id);
CREATE INDEX IF NOT EXISTS idx_favorites_user_order ON favorites(user_id, display_order);


-- ============================================================
-- 4. USER SETTINGS (Synced app preferences)
-- ============================================================
-- Stores all user-configurable settings that should sync across devices.
-- Uses a single row per user (JSONB would work too, but explicit columns
-- are easier to validate and query).
--
-- The iOS app stores these locally in @AppStorage and syncs on login.
-- This is the source of truth for cross-device consistency.

CREATE TABLE IF NOT EXISTS user_settings (
  user_id UUID REFERENCES auth.users ON DELETE CASCADE PRIMARY KEY,
  
  -- Appearance
  preferred_theme TEXT DEFAULT 'system' CHECK (preferred_theme IN ('system', 'light', 'dark')),
  
  -- Transit search radii (meters)
  near_you_radius_meters DOUBLE PRECISION DEFAULT 2414,       -- ~1.5 mi
  farther_away_radius_meters DOUBLE PRECISION DEFAULT 4023,   -- ~2.5 mi
  much_farther_away_radius_meters DOUBLE PRECISION DEFAULT 8047, -- ~5.0 mi
  
  -- Map & display
  show_system_map BOOLEAN DEFAULT true,                       -- Show full transit system map overlay
  subway_line_offset_meters DOUBLE PRECISION DEFAULT 12,      -- How far apart co-located subway lines spread
  
  -- General behavior
  haptics_enabled BOOLEAN DEFAULT true,                       -- Haptic feedback on interactions
  auto_refresh_enabled BOOLEAN DEFAULT true,                  -- Auto-refresh transit data on timer
  notifications_enabled BOOLEAN DEFAULT true,                 -- Push notifications
  
  -- Developer (not typically synced, but useful for multi-device dev)
  dev_use_localhost BOOLEAN DEFAULT false,
  dev_custom_ip TEXT DEFAULT '',
  
  -- Metadata
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);

-- Trigger for auto-updating updated_at (idempotent)
DROP TRIGGER IF EXISTS update_user_settings_updated_at ON user_settings;
CREATE TRIGGER update_user_settings_updated_at
    BEFORE UPDATE ON user_settings
    FOR EACH ROW
    EXECUTE FUNCTION update_updated_at_column();

-- Auto-create default settings row when a new user signs up
CREATE OR REPLACE FUNCTION handle_new_user_settings()
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO public.user_settings (user_id)
  VALUES (NEW.id)
  ON CONFLICT (user_id) DO NOTHING;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

DROP TRIGGER IF EXISTS on_auth_user_created_settings ON auth.users;
CREATE TRIGGER on_auth_user_created_settings
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE FUNCTION handle_new_user_settings();


-- ============================================================
-- 5. WIDGET SCHEDULES (When should the widget activate)
-- ============================================================
-- Stores when users want their widgets to automatically show transit.
-- Syncs across devices so schedules work on iPad, iPhone, etc.
-- NOTE: Section numbers 3→5 shifted due to new favorites + user_settings tables.
--
-- IMPORTANT: days_of_week uses 0-based indexing to match Swift code:
--   Swift: calendar.component(.weekday) returns 1-7
--   WidgetSchedule.swift converts to 0-based: weekday - 1
--   So: 0=Sunday, 1=Monday, 2=Tuesday, ..., 6=Saturday

CREATE TABLE IF NOT EXISTS schedules (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES auth.users ON DELETE CASCADE NOT NULL,
  
  -- Schedule config
  days_of_week INT[] NOT NULL,         -- 0=Sunday, 1=Monday, ..., 6=Saturday
  start_time TIME NOT NULL,            -- e.g., '08:30:00'
  duration_minutes INT DEFAULT 15,     -- How long the widget stays active
  
  -- Optional: tie to a specific route (future feature)
  route_id TEXT,
  direction TEXT,
  
  -- State
  is_enabled BOOLEAN DEFAULT true,
  
  -- Metadata
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);

-- Trigger for auto-updating updated_at (idempotent)
DROP TRIGGER IF EXISTS update_schedules_updated_at ON schedules;
CREATE TRIGGER update_schedules_updated_at
    BEFORE UPDATE ON schedules
    FOR EACH ROW
    EXECUTE FUNCTION update_updated_at_column();


-- ============================================================
-- 6. COMMUTE PATTERNS (Smart Suggestions)
-- ============================================================
-- Learns user's commute patterns based on:
-- - Location where they usually check the app
-- - Time of day they travel
-- - Routes they typically take
--
-- Used by SmartSuggester to predict what route you probably want.
--
-- IMPORTANT: day_of_week uses Swift's Calendar.component(.weekday) DIRECTLY:
--   1=Sunday, 2=Monday, 3=Tuesday, ..., 7=Saturday
--   (Different from schedules table which uses 0-based!)

CREATE TABLE IF NOT EXISTS commute_patterns (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES auth.users ON DELETE CASCADE NOT NULL,
  
  -- Pattern data
  route_id TEXT NOT NULL,
  direction TEXT NOT NULL,
  start_latitude DOUBLE PRECISION NOT NULL,
  start_longitude DOUBLE PRECISION NOT NULL,
  destination_station_id TEXT NOT NULL,
  destination_name TEXT NOT NULL,
  
  -- Time context
  time_of_day INT NOT NULL CHECK (time_of_day >= 0 AND time_of_day <= 23),  -- Hour 0-23
  day_of_week INT NOT NULL CHECK (day_of_week >= 1 AND day_of_week <= 7),   -- 1=Sunday, 7=Saturday
  
  -- Frequency tracking
  frequency INT DEFAULT 1,             -- How often this pattern occurs
  last_used TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()),
  
  -- Metadata
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()),
  
  -- Unique constraint to avoid duplicates
  UNIQUE(user_id, route_id, direction, destination_station_id, time_of_day, day_of_week)
);

-- Trigger for auto-updating updated_at (idempotent)
DROP TRIGGER IF EXISTS update_commute_patterns_updated_at ON commute_patterns;
CREATE TRIGGER update_commute_patterns_updated_at
    BEFORE UPDATE ON commute_patterns
    FOR EACH ROW
    EXECUTE FUNCTION update_updated_at_column();

DROP POLICY IF EXISTS "Users can update own profile" ON profiles;
DROP POLICY IF EXISTS "Users can manage own favorites" ON favorites;
DROP POLICY IF EXISTS "Users can manage own settings" ON user_settings;
DROP POLICY IF EXISTS "Users can manage own schedules" ON schedules;

-- PROFILES: Users can only see/edit their own profile
CREATE POLICY "Users can view own profile" 
  ON profiles FOR SELECT 
  USING (auth.uid() = id);

CREATE POLICY "Users can update own profile" 
  ON profiles FOR UPDATE 
  USING (auth.uid() = id);

-- FAVORITES: Users can only manage their own favorites
CREATE POLICY "Users can manage own favorites" 
  ON favorites FOR ALL 
  USING (auth.uid() = user_id);

-- USER SETTINGS: Users can only manage their own settings
CREATE POLICY "Users can manage own settings" 
  ON user_settings FOR ALL 
  USING (auth.uid() = user_id);

-- SCHEDULES: Users can only manage their own schedules
CREATE POLICY "Users can manage own schedules" 
  ON schedules FOR ALL 
  USING (auth.uid() = user_id);

-- COMMUTE PATTERNS: Users can only manage their own patterns
CREATE POLICY "Users can manage own commute patterns" 
  ON commute_patterns FOR ALL 
  USING (auth.uid() = user_id);

-- ROUTE INTERACTIONS: Anyone can insert (for analytics), users can view their own
CREATE POLICY "Anyone can insert route interactions" 
  ON route_interactions FOR INSERT 
  WITH CHECK (true);

CREATE POLICY "Users can view own interactions" 
  ON route_interactions FOR SELECT 
  USING (auth.uid() = user_id OR user_id IS NULL);

-- Get most popular routes (for backend trending feature)
CREATE OR REPLACE FUNCTION get_popular_routes(
  p_limit INT DEFAULT 10,
  p_mode TEXT DEFAULT NULL,
  p_days INT DEFAULT 7
)
RETURNS TABLE (
  route_id TEXT,
  route_display_name TEXT,
  mode TEXT,
  interaction_count BIGINT
) AS $$
BEGIN
  RETURN QUERY
  SELECT 
    ri.route_id,
    MAX(ri.route_display_name) as route_display_name,
    ri.mode,
    COUNT(*) as interaction_count
  FROM route_interactions ri
  WHERE ri.created_at >= NOW() - (p_days || ' days')::INTERVAL
    AND (p_mode IS NULL OR ri.mode = p_mode)
  GROUP BY ri.route_id, ri.mode
  ORDER BY interaction_count DESC
  LIMIT p_limit;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;


-- Auto-create profile when user signs up
CREATE OR REPLACE FUNCTION handle_new_user()
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO public.profiles (id, email, created_at)
  VALUES (NEW.id, NEW.email, NOW())
  ON CONFLICT (id) DO NOTHING;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Trigger to auto-create profile on signup (idempotent)
DROP TRIGGER IF EXISTS on_auth_user_created ON auth.users;
CREATE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE FUNCTION handle_new_user();
